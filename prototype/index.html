# hợp nhất kết quả từ semantic search và BM25
def hybrid_search(query, k=10):
    dense  = semantic_search(query, k)      # theo nghĩa
    sparse = bm25_search(query, k)          # theo từ khoá
    return reciprocal_rank_fusion(dense, sparse, k)
